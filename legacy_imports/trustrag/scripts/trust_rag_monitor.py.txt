#!/usr/bin/env python3
"""
TrustRAG 2.0 监控仪表板 (Monitoring Dashboard)

整合所有监控系统的统一界面：
- Token成本监控
- 召回率改进监控
- 系统健康检查
- 升级进度跟踪

使用方法：
python scripts/trust_rag_monitor.py --show all
python scripts/trust_rag_monitor.py --alerts
python scripts/trust_rag_monitor.py --export-report
"""
import os
import sys
import json
import argparse
from pathlib import Path
from datetime import datetime
from typing import Dict, List, Any

# Add project root to path
project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))

from trust_rag.core.monitoring.token_cost_tracker import get_token_tracker
from trust_rag.core.monitoring.recall_monitor import get_recall_monitor
from trust_rag.runtime.dispatcher.celery_queue import get_celery_app
from trust_rag.core.judgment.shadow_arbitrator import ArbitrationMode


class TrustRAGMonitor:
    """
    TrustRAG 2.0 监控仪表板
    """

    def __init__(self):
        self.token_tracker = get_token_tracker()
        self.recall_monitor = get_recall_monitor()

    def get_system_health(self) -> Dict[str, Any]:
        """获取系统整体健康状态"""
        health = {
            "timestamp": datetime.now().isoformat(),
            "overall_status": "UNKNOWN",
            "components": {},
            "alerts": [],
            "metrics": {}
        }

        try:
            # Token成本监控
            cost_metrics = self.token_tracker.get_cost_metrics()
            health["components"]["token_cost"] = {
                "status": "healthy" if cost_metrics.total_cost_usd < 50 else "warning",
                "daily_cost": cost_metrics.total_cost_usd,
                "queries_today": cost_metrics.queries_count
            }

            # 召回率监控
            recall_report = self.recall_monitor.export_monitoring_report()
            health["components"]["recall_quality"] = {
                "status": "healthy" if recall_report.get("overall_status") == "HEALTHY" else "warning",
                "recall_at_5": recall_report["current_metrics"]["recall_at_5"],
                "improvement": recall_report.get("algorithm_comparison", {}).get("improvements", {}).get("recall_at_5", 0)
            }

            # 影子仲裁状态
            try:
                from trust_rag.core.judgment.shadow_arbitrator import get_shadow_arbitrator
                arbitrator = get_shadow_arbitrator()
                stats = arbitrator.get_stats()
                health["components"]["shadow_arbitration"] = {
                    "status": "healthy",
                    "mode": stats["mode"],
                    "total_queries": stats["total_queries"],
                    "new_used": stats["new_used"]
                }
            except:
                health["components"]["shadow_arbitration"] = {
                    "status": "not_deployed",
                    "mode": "none"
                }

            # 异步队列状态
            try:
                celery_app = get_celery_app()
                queue_stats = celery_app.get_queue_stats()
                health["components"]["async_queue"] = {
                    "status": "healthy" if queue_stats.get("total_pending", 0) < 100 else "warning",
                    "pending_tasks": queue_stats.get("pending_count", 0),
                    "workers": len(queue_stats.get("active_workers", []))
                }
            except:
                health["components"]["async_queue"] = {
                    "status": "not_deployed",
                    "pending_tasks": 0
                }

            # 综合状态判断
            component_statuses = [comp["status"] for comp in health["components"].values()]
            if "critical" in component_statuses:
                health["overall_status"] = "CRITICAL"
            elif "warning" in component_statuses:
                health["overall_status"] = "WARNING"
            elif all(s == "healthy" for s in component_statuses):
                health["overall_status"] = "HEALTHY"
            else:
                health["overall_status"] = "DEGRADED"

            # 生成告警
            health["alerts"] = self._generate_alerts(health)

            # 关键指标
            health["metrics"] = {
                "cost_savings": self._calculate_cost_savings(),
                "recall_improvement": self._calculate_recall_improvement(),
                "system_uptime": self._estimate_uptime(),
                "target_achievement": self._check_targets_achievement()
            }

        except Exception as e:
            health["overall_status"] = "ERROR"
            health["error"] = str(e)

        return health

        # 关键指标
        print("📈 关键指标:")
        metrics = health.get("metrics", {})

        for stage_num in [1, 2, 3]:

    def show_cost_analysis(self):
        """显示Token成本分析"""
        print("💰 Token成本分析")
        print("=" * 40)

        # 总体成本指标
        cost_metrics = self.token_tracker.get_cost_metrics()
        print("📊 总体成本指标 (24小时):"        print(".4f"        print(f"📈 总Token消耗: {cost_metrics.total_tokens:,}")
        print(".4f"        print(".2f"        print()

        # 分层解析效率
        parsing_report = self.token_tracker.get_parsing_efficiency_report()
        print("🔍 分层解析效率:")
        tiers = parsing_report.get("parsing_tiers", {})
        for tier_name, stats in tiers.items():
            if stats.get("documents_processed", 0) > 0:
                savings = stats.get("token_savings_percent", 0)
                savings_icon = "💚" if savings >= 60 else "💛" if savings >= 30 else "❤️"
                print(".1f" f"  {savings_icon} {tier_name}: {stats['documents_processed']} docs, "
                      f"${stats['avg_cost_per_doc']:.4f}/doc, 节省{savings:.1f}%")

        efficiency = parsing_report.get("overall_efficiency", {})
        if efficiency:
            target_status = efficiency.get("target_achievement", "")
            print(f"🎯 总体效率: 轻量级占比{efficiency.get('lightweight_ratio', 0):.1%}, "
                  f"加权成本${efficiency.get('weighted_avg_cost_per_doc', 0):.4f}/doc")
            print(f"📈 {target_status}")
        print()

        # 成本预测
        predictions = self.token_tracker.get_cost_predictions()
        if "error" not in predictions:
            print("🔮 成本预测 (未来7天):")
            for pred in predictions.get("predictions", []):
                print(f"  Day {pred['day']}: ${pred['predicted_daily_cost']:.2f} "
                      f"(累计: ${pred['predicted_cumulative']:.2f})")
        print()

        # 预算检查
        budget_status = self.token_tracker.check_budget_limits()
        print("💳 预算状态:")
        print(".4f"        print(".4f"        print(".1%"        alerts = budget_status.get("alerts", [])
        for alert in alerts:
            print(f"  🚨 {alert['level']}: {alert['message']}")

        print("🎯 召回率改进分析")
        print("=" * 40)

        # 当前指标
        monitor = self.recall_monitor
        print("📊 当前检索指标:"        print(".1%"        print(".1%"        print(".1%"        print(".1%"        print(".1%"        print()

        # 与基准对比
        comparison = monitor.calculate_improvement_metrics()
        print("⚖️ 与基准算法对比:")
        for metric, improvement in comparison.improvement.items():
            icon = "📈" if improvement > 0 else "📉" if improvement < 0 else "➡️"

        bge_validation = monitor.validate_bge_improvement()

            print("🧠 BGE-M3 + 重排序验证:")
            for metric, result in bge_validation.get("validation_results", {}).items():

            print()

    def export_full_report(self, output_file: str = None):
        """导出完整监控报告"""
        if not output_file:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            output_file = f"artifacts/monitoring/full_report_{timestamp}.json"

        # 收集所有监控数据
        report = {
            "timestamp": datetime.now().isoformat(),
            "system_health": self.get_system_health(),
            "token_cost_analysis": self.token_tracker.export_metrics_report(),
            "recall_quality_analysis": self.recall_monitor.export_monitoring_report(),
            "rollout_status": self._get_rollout_status()
        }

        # 确保输出目录存在
        Path(output_file).parent.mkdir(parents=True, exist_ok=True)

        # 导出JSON报告
        with open(output_file, 'w', encoding='utf-8') as f:
            json.dump(report, f, indent=2, ensure_ascii=False, default=str)

        print(f"📄 完整监控报告已导出到: {output_file}")
        return output_file

    def _status_icon(self, status: str) -> str:
        """状态图标"""
        icons = {
            "HEALTHY": "🟢",
            "WARNING": "🟡",
            "CRITICAL": "🔴",
            "ERROR": "💥",
            "UNKNOWN": "❓",
            "DEGRADED": "🟠"
        }
        return icons.get(status.upper(), "❓")

    def _alert_icon(self, level: str) -> str:
        """告警图标"""
        icons = {
            "CRITICAL": "🚨",
            "WARNING": "⚠️",
            "INFO": "ℹ️"
        }
        return icons.get(level.upper(), "❓")

    def _component_display_name(self, comp_name: str) -> str:
        """组件显示名称"""
        names = {
            "token_cost": "Token成本监控",
            "recall_quality": "召回率质量",
            "shadow_arbitration": "影子仲裁",
            "async_queue": "异步队列"
        }
        return names.get(comp_name, comp_name)

    def _generate_alerts(self, health: Dict[str, Any]) -> List[Dict[str, Any]]:
        """生成告警列表"""
        alerts = []

        # Token成本告警
        cost_info = health["components"].get("token_cost", {})
        if cost_info.get("status") == "warning":
            alerts.append({
                "level": "WARNING",
                "component": "token_cost",
                "message": ".2f"            })

        # 召回率告警
        recall_info = health["components"].get("recall_quality", {})
        if recall_info.get("status") == "warning":
            alerts.append({
                "level": "WARNING",
                "component": "recall_quality",
                "message": ".1%"            })

        # 队列积压告警
        queue_info = health["components"].get("async_queue", {})
        if queue_info.get("pending_tasks", 0) > 100:
            alerts.append({
                "level": "CRITICAL",
                "component": "async_queue",
                "message": f"异步队列积压严重: {queue_info['pending_tasks']} 个待处理任务"
            })

        return alerts

    def _calculate_cost_savings(self) -> Dict[str, Any]:
        """计算成本节省"""
        parsing_report = self.token_tracker.get_parsing_efficiency_report()
        efficiency = parsing_report.get("overall_efficiency", {})

        if not efficiency:
            return {"savings_percent": 0, "message": "数据不足"}

        weighted_cost = efficiency.get("weighted_avg_cost_per_doc", 0)
        traditional_cost = 0.15  # 假设传统方法0.15美元/文档

        if traditional_cost > 0:
            savings_percent = (1 - weighted_cost / traditional_cost) * 100
            return {
                "savings_percent": round(savings_percent, 1),
                "message": f"相比传统解析节省{savings_percent:.1f}% Token成本"
            }

        return {"savings_percent": 0, "message": "无法计算节省"}

    def _calculate_recall_improvement(self) -> Dict[str, Any]:
        """计算召回率改进"""
        comparison = self.recall_monitor.calculate_improvement_metrics()
        recall_improvement = comparison.improvement.get("recall_at_5", 0)

        return {
            "improvement_percent": recall_improvement,
            "message": f"Recall@5 提升{recall_improvement:.1f}%" if recall_improvement > 0
                     else f"Recall@5 下降{abs(recall_improvement):.1f}%" if recall_improvement < 0
                     else "Recall@5 无显著变化"
        }

    def _estimate_uptime(self) -> float:
        """估算系统正常运行时间"""
        # 这里可以集成实际的监控数据
        # 暂时返回一个估算值
        return 99.5

    def _check_targets_achievement(self) -> Dict[str, Any]:
        """检查目标达成情况"""
        # 检查主要目标：Token成本降低60%，召回率提升至90%

        cost_savings = self._calculate_cost_savings()
        recall_improvement = self._calculate_recall_improvement()

        cost_target_achieved = cost_savings.get("savings_percent", 0) >= 60
        recall_target_achieved = recall_improvement.get("improvement_percent", 0) >= 0  # 至少不下降

        if cost_target_achieved and recall_target_achieved:
            return {"status": "HEALTHY", "message": "✅ 主要目标已达成"}
        elif cost_target_achieved:
            return {"status": "WARNING", "message": "⚠️ Token成本目标达成，但召回率需要改进"}
        elif recall_target_achieved:
            return {"status": "WARNING", "message": "⚠️ 召回率目标达成，但Token成本节省不足"}
        else:
            return {"status": "CRITICAL", "message": "❌ 主要目标均未达成"}

    def _get_rollout_status(self) -> Dict[str, Any]:
        """获取上线状态"""
        # 这里可以读取rollout日志
        # 暂时返回模拟状态
        return {
            "stage_1": {"completed": True, "completion_rate": 1.0},
            "stage_2": {"in_progress": True, "completion_rate": 0.6},
            "stage_3": {"completed": False, "completion_rate": 0.0}
        }


def main():
    """主入口函数"""
    parser = argparse.ArgumentParser(description="TrustRAG 2.0 监控仪表板")
    parser.add_argument("--show", choices=["dashboard", "cost", "recall", "all"],
                       default="dashboard", help="显示内容")
    parser.add_argument("--alerts", action="store_true", help="只显示告警")
    parser.add_argument("--export-report", type=str, nargs='?', const=True,
                       help="导出完整报告 (可选指定输出文件)")

    args = parser.parse_args()

    monitor = TrustRAGMonitor()

    if args.export_report:
        if isinstance(args.export_report, str):
            output_file = args.export_report
        else:
            output_file = None
        monitor.export_full_report(output_file)
        return

    if args.alerts:
        health = monitor.get_system_health()
        alerts = health.get("alerts", [])
        if alerts:
            print("🚨 活动告警:")
            for alert in alerts:
                print(f"  {monitor._alert_icon(alert['level'])} {alert['level']}: {alert['message']}")
        else:
            print("✅ 无活动告警")
        return

    # 显示请求的内容
    if args.show == "dashboard" or args.show == "all":
        monitor.show_dashboard()
        if args.show == "all":
            print("\n" + "="*60 + "\n")

    if args.show == "cost" or args.show == "all":
        monitor.show_cost_analysis()
        if args.show == "all":
            print("\n" + "="*60 + "\n")

    if args.show == "recall" or args.show == "all":
        monitor.show_recall_analysis()


if __name__ == "__main__":
    main()
