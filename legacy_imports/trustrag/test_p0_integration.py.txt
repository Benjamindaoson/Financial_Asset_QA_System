def test_document_ingestion():

            print("  ✅ 向量索引更新成功")

def test_vector_retrieval():
    """测试向量检索"""
    print("\n🔍 测试向量检索...")

    try:
        from trust_rag.system import TrustRAG

        system = TrustRAG()

        # 测试查询
        test_query = "What is NVIDIA's revenue for FY2023?"

        start_time = time.time()
        result = system.process_query(test_query)
        latency = time.time() - start_time

        print("  ✅ 查询处理成功"        print(f"    查询: {test_query}")
        print(f"    响应时间: {latency:.2f}s")
        print(f"    判决: {result.verdict}")
        print(f"    证据数量: {len(result.evidence)}")

        if result.answer:
            print(f"    答案: {result.answer.text[:100]}...")
            print(f"    置信度: {result.answer.confidence}")

        # 检查是否使用了向量检索
        verification_data = result.verification_data
        if verification_data.get("retrieval_error"):
            print("  ❌ 检索失败 - 向量索引可能未正确配置")
            return False
        else:
            print("  ✅ 检索成功 - 向量索引工作正常")
            return True

    except Exception as e:
        print(f"  ❌ 向量检索测试失败: {e}")
        return False

def main():
    """主测试函数"""
    print("🚀 TrustRAG 2.0 P0 Critical Issues Integration Test")
    print("=" * 60)
    print("测试目标: 验证三个核心P0问题的修复")
    print("1. ✅ RetrievalAdapter集成BGE-M3和pgvector")
    print("2. ✅ eval/harness.py语法错误修复")
    print("3. ✅ 从JSON/FAISS迁移到PostgreSQL+pgvector")
    print()

    tests = [
        ("系统初始化", test_system_initialization),
        ("文档摄入", test_document_ingestion),
        ("向量检索", test_vector_retrieval),
        ("评估框架", test_evaluation_framework),
        ("存储迁移", test_storage_migration)
    ]

    results = []
    for test_name, test_func in tests:
        try:
            result = test_func()
            results.append((test_name, result))
        except Exception as e:
            print(f"❌ {test_name} 测试异常: {e}")
            results.append((test_name, False))

    # 汇总结果
    print("\n" + "=" * 60)
    print("📊 测试结果汇总")

    passed = 0
    total = len(results)

    for test_name, result in results:
        status = "✅ 通过" if result else "❌ 失败"
        print(f"  {test_name}: {status}")
        if result:
            passed += 1

    success_rate = passed / total if total > 0 else 0
    print(".1%")

    # 详细分析
    print("\n🔍 详细分析:")

    if success_rate >= 0.8:
        print("🎉 核心P0问题已基本解决！系统具备生产就绪的基础。")
        print("📋 剩余工作:")
        print("  - 配置PostgreSQL数据库连接")
        print("  - 下载BGE-M3模型文件")
        print("  - 准备测试数据集")
        print("  - 执行完整的端到端验证")
        return 0
    else:
        print("⚠️  仍存在关键问题需要解决:")
        failed_tests = [name for name, result in results if not result]
        for test in failed_tests:
            print(f"  - {test}")
        return 1

if __name__ == "__main__":
    exit_code = main()
    sys.exit(exit_code)
